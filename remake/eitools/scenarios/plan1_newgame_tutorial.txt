# План 1: новая игра, вход в зону, игровое меню и выход.
# Главное меню -> New game -> Normal -> ролик -> зона1 (подсказки обучения выключены в обеих играх,
# страницы обучения больше не листаются). Затем Esc открывает игровое меню,
# Exit to main menu возвращает в главное меню, EXIT закрывает игру.
# Строки sync держат обе игры вместе: каждая ждёт другую перед следующим шагом.
# Отчёт и найденные различия: plan1_newgame_tutorial.md

# потолок времени на весь прогон; при его превышении игры останавливаются, прогон неудачен
timeout 90000

# настройки обеих игр сбрасываются к встроенным умолчаниям перед прогоном (копия в run/settings_backup)
reset_settings

# главное меню: ждём, пока камень NEW GAME станет светлым (загрузка тёмная);
# камень бывает подсвечен, поэтому допуск цвета больше, чем в common_intro
waitpixel 430 200 128,123,92 70 30000
# даём окну и меню устояться после расстановки окон
wait 1000
sync
shot 01_menu

# курсор над NEW GAME: подсветка пункта меню
sync
move 425 195
wait 800
sync
shot 02_menu_hover

# New game -> окно сложности
sync
click 425 195
wait 1200
sync
shot 03_difficulty

# Normal
sync
click 297 482

# подсказки обучения выключены в обеих играх (демо без --tutorial, оригинал с ShowTutorial=0 от eidrive):
# у оригинала один Space останавливает ролик (второй Space в мире поставил бы паузу), у демо ролика нет.
original: wait 2000
original: key Space
original: wait 1000
demo: wait 3000
# зона загружена, когда в окне информации видна зелёная полоса здоровья героя
# (оригинал 16,198,0, демо 8,161,0)
waitpixel 90 192 12,180,0 45 30000
wait 1500
sync
shot 04_zone

# игровое меню Esc
sync
key Escape
wait 1200
sync
shot 10_esc_menu

# Exit to main menu: обе игры спрашивают "Quit game", подтверждает галочка (300, 375)
sync
click 400 405
wait 1500
sync
shot 11_quit_confirm
click 300 375
waitpixel 430 200 128,123,92 70 30000
wait 1000
sync
shot 12_menu_back

# EXIT главного меню закрывает игру
sync
click 425 410
wait 2000

# игра закрывается сама
expect_closed 10000
