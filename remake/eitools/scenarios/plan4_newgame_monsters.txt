# Сценарий 3: новая игра, закрыть обучение и подойти к трём людям у старта.
# Запуск: обе игры уже в главном меню (compare перезапускает их сам).
# В zone1.mob три человека стоят восточнее героя (x≈75..78 против x≈54),
# сразу за начальной камерой; поэтому идём кликами у нижнего края экрана.
# Кликаем правее портрета героя (450,550), чтобы не попасть в HUD.
# Строки sync держат шаги обеих игр вместе; в конце compare остановит игры сам.

# потолок времени на весь прогон; при его превышении игры останавливаются, прогон неудачен
timeout 90000

# настройки обеих игр сбрасываются к умолчаниям перед прогоном (копия в run/settings_backup)
reset_settings

# главное меню
waitpixel 430 200 128,123,92 70 30000
wait 1000
sync
shot 01_menu

# New game -> окно сложности -> Normal
sync
click 425 195
wait 1200
sync
shot 02_difficulty

sync
click 297 482

demo: wait 3000

waitpixel 90 192 12,180,0 45 30000

# люди восточнее героя, а камера смотрит на север: камера едет на восток стрелкой Right
# (Camera right у обеих игр). Поворот правой кнопкой не годится: у game.exe скорость поворота
# задаёт отвод курсора от центра экрана, пока кнопка нажата (5eabd0, 40f360), у демо - сдвиг мыши (P15)
sync
keydown Right
wait 1500
keyup Right
wait 1500
sync
shot 04b_east
